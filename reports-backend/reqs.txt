pipeline {
    agent any

    tools {
        maven 'MAVEN-HOME'
    }

    stages {
        stage('Git Repo & Clean') {
            steps {
                bat "git clone YOUR-GITHUB-REPO-LINK"
                bat "mvn clean -f PROJECT-FOLDER"
            }
        }

        stage('Install') {
            steps {
                bat "mvn install -f PROJECT-FOLDER"
            }
        }

        stage('Test') {
            steps {
                bat "mvn test -f PROJECT-FOLDER"
            }
        }

        stage('Package') {
            steps {
                bat "mvn package -f PROJECT-FOLDER"
            }
        }
    }
}
